Table users {
  id               int          [pk, increment]
  username         varchar(50)  [not null, unique]
  email            varchar(255) [not null, unique]
  password_hash    varchar(255) [not null]
  is_active        tinyint(1)   [not null, default: 1]
  flag_count       int          [not null, default: 0, note: 'Opgehoogd bij flags die meetellen voor review']
  is_under_review  tinyint(1)   [not null, default: 0, note: 'Auto-true bij drempel, manager-queue']
  deleted_at       datetime     [note: 'Soft delete: gevuld = geanonimiseerd']
  created_at       datetime     [not null, default: `CURRENT_TIMESTAMP`]
  updated_at       datetime     [not null, default: `CURRENT_TIMESTAMP`]
}

Table districts {
  id               int          [pk, increment]
  name             varchar(100) [not null, unique]
  postal_prefix    varchar(10)  [note: 'Bv. 3241 voor Middelharnis']
  center_lat       decimal(10,8)
  center_lng       decimal(11,8)
  radius_meters    int          [note: 'Straal voor automatische toewijzing']
  is_active        tinyint(1)   [not null, default: 1]
  created_at       datetime     [not null, default: `CURRENT_TIMESTAMP`]
  updated_at       datetime     [not null, default: `CURRENT_TIMESTAMP`]
}

Table officers {
  id               int          [pk, increment]
  username         varchar(50)  [not null, unique]
  email            varchar(255) [not null, unique]
  password_hash    varchar(255) [not null]
  badge_number     varchar(20)  [not null, unique]
  // Wijken via district_officer; een BOA kan aan meerdere wijken gekoppeld zijn.
  // Afdelingen via department_officer; een BOA kan aan meerdere afdelingen gekoppeld zijn.
  is_active        tinyint(1)   [not null, default: 1]
  deleted_at       datetime
  created_at       datetime     [not null, default: `CURRENT_TIMESTAMP`]
  updated_at       datetime     [not null, default: `CURRENT_TIMESTAMP`]
}

Table managers {
  id               int          [pk, increment]
  username         varchar(50)  [not null, unique]
  email            varchar(255) [not null, unique]
  password_hash    varchar(255) [not null]
  // Afdelingen via department_manager; een manager kan aan meerdere afdelingen gekoppeld zijn.
  // Wijken via district_manager; een manager kan aan meerdere wijken gekoppeld zijn.
  // is_main_manager: alleen via seeding/admin; nooit via publieke API. Route-niveau (geen extra kolommen):
  //   GET/PATCH/PATCH disable /api/main-managers — alleen actieve main manager; binding op is_main_manager = true.
  //   GET/POST/PATCH/PATCH disable /api/managers — alleen actieve main manager; binding op is_main_manager = false.
  is_main_manager  boolean      [not null, default: false, note: 'Hiërarchievlag; API mag niet wijzigen; route binding filtert main vs gewone managers']
  created_by_manager_id int     [ref: > managers.id, note: 'Optioneel; wie deze manager heeft aangemaakt']
  is_active        tinyint(1)   [not null, default: 1]
  deleted_at       datetime
  created_at       datetime     [not null, default: `CURRENT_TIMESTAMP`]
  updated_at       datetime
}

Table issues {
  id                   int          [pk, increment]
  user_id              int          [ref: > users.id, note: 'NULL na soft delete user']
  category_id          int          [not null, ref: > categories.id]
  assigned_officer_id  int          [ref: > officers.id]
  district_id          int          [not null, ref: > districts.id, note: 'Verplichte singuliere issue-locatie/referentie; restrict delete; valt buiten actor-wijk many-to-many toewijzingen']
  duplicate_of_id      int          [ref: > issues.id]
  title                varchar(255) [not null]
  content              text         [not null]
  postal_code          varchar(10)
  address              varchar(255)
  latitude             decimal(10,8)
  longitude            decimal(11,8)
  status               varchar(20)  [not null, default: 'open', note: 'open | in_behandeling | afgerond | afgewezen | gedupliceerd']
  priority             tinyint      [note: 'unsigned nullable; server-afgeleid uit hoofdcategorie priority (lager getal = hogere urgentie); client mag niet schrijven']
  duplicate_count      int          [not null, default: 0]
  participant_count    int          [not null, default: 0, note: 'Direct opgehoogd, nachtelijks gereconcilieerd']
  is_flagged           tinyint(1)   [not null, default: 0]
  visibility           varchar(20)  [not null, default: 'visible', note: 'visible | hidden']
  is_anonymous         tinyint(1)   [not null, default: 0, note: 'User_id blijft eigenaar; publiek + BOA zien alleen alias']
  anonymous_alias      varchar(20)  [note: 'Bv. Melder#A1B2C3D4, stabiel server-generated alias getoond i.p.v. username; unique index (meerdere NULL toegestaan)']
  chat_status          varchar(10)  [not null, default: 'closed', note: 'open | closed - alleen BOA bedient dit']
  chat_closed_by_officer_id int     [ref: > officers.id]
  resolved_at          datetime
  created_at           datetime     [not null, default: `CURRENT_TIMESTAMP`]
  updated_at           datetime     [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    anonymous_alias [unique, note: 'issues_anonymous_alias_unique; meerdere NULL toegestaan']
  }
}
